"""Public worker frame.close queued immediately before owner carrier.aclose."""
import argparse
import asyncio
import gc
import hashlib
import json
from pathlib import Path
import socket
import sys
import threading
import traceback

p=argparse.ArgumentParser(description=__doc__)
p.add_argument('--mesh',type=Path,required=True);p.add_argument('--output',type=Path,required=True)
a=p.parse_args();a.mesh=a.mesh.resolve();sys.path.insert(0,str(a.mesh/'src'));gc.disable()
from mesh_plus.sync import control_carrier_from_ipc
from mesh_plus.sync.budget import AllocationLedger
from mesh_plus.transport import boottime_ms

async def main():
    loop=asyncio.get_running_loop();previous=loop.get_exception_handler();errors=[]
    def capture(_loop,context):errors.append(context.get('exception'))
    loop.set_exception_handler(capture)
    ledger=AllocationLedger();local,peer=socket.socketpair();sibling_local,sibling_peer=socket.socketpair()
    carrier=control_carrier_from_ipc(local,ledger=ledger)
    sibling=control_carrier_from_ipc(sibling_local,ledger=ledger)
    peer.sendall(b'{"payload":"retire without a materialize pin"}\n')
    frame=await carrier.receive_control(maximum_bytes=8192,deadline_ms=boottime_ms()+2000)
    reserved=ledger.reserved
    # The short join deliberately prevents the owner-loop callback from running
    # until public aclose begins. No private source state or hooks are mutated.
    worker=threading.Thread(target=frame.close);worker.start();worker.join(2)
    assert not worker.is_alive()
    await carrier.aclose()
    sibling_peer.sendall(b'{}\n')
    other=await sibling.receive_control(maximum_bytes=8192,deadline_ms=boottime_ms()+2000)
    other.close();await sibling.aclose()
    frame.close();await carrier.aclose()
    await asyncio.sleep(0)
    counters=[ledger.used,ledger.reserved,ledger.encoded,ledger.encoded_reserved,ledger.pending_units]
    diagnostics=[{'type':type(e).__name__,'message':str(e),'trace':''.join(traceback.format_exception(e))} for e in errors]
    peer.close();sibling_peer.close();loop.set_exception_handler(previous)
    loaded={}
    for module in tuple(sys.modules.values()):
        file=getattr(module,'__file__',None)
        if file and Path(file).suffix=='.py' and Path(file).resolve().is_relative_to(a.mesh):
            path=Path(file).resolve();loaded[str(path.relative_to(a.mesh))]=hashlib.sha256(path.read_bytes()).hexdigest()
    report={'python':sys.version.split()[0],'gcDisabled':not gc.isenabled(),'heapBeforeWorkerClose':reserved,
            'finalCounters':counters,'callbackErrors':diagnostics,'siblingUsable':True,'loadedInputs':loaded,
            'scope':'Public minted carrier/frame/close and ledger APIs; real worker thread and bounded join control ordering. No source hooks or private source mutations, no wire/authority/RSS/cap-overrun claim. Actual callback exceptions retained with GC disabled.'}
    a.output.write_text(json.dumps(report,indent=2)+'\n')
    print(json.dumps({k:report[k] for k in ('python','heapBeforeWorkerClose','finalCounters','siblingUsable')}|{'callbackErrors':[(x['type'],x['message']) for x in diagnostics],'loadedPins':len(loaded)}))

asyncio.run(main())
