"""Controlled copy/close interleavings through public carrier/frame APIs."""
import argparse
import asyncio
import copy
import gc
import hashlib
import json
from pathlib import Path
import socket
import sys
import threading
from unittest.mock import patch

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--mesh', type=Path, required=True)
parser.add_argument('--output', type=Path, required=True)
args = parser.parse_args()
args.mesh = args.mesh.resolve()
sys.path.insert(0, str(args.mesh / 'src'))
gc.disable()
from mesh_plus.protocol import ProtocolError
from mesh_plus.sync import control_carrier_from_ipc
from mesh_plus.sync.budget import AllocationLedger
from mesh_plus.transport import boottime_ms

async def main():
    cases = []
    for mode in ('reentrant-frame-close', 'concurrent-carrier-close'):
        ledger = AllocationLedger()
        local, peer = socket.socketpair()
        sibling_local, sibling_peer = socket.socketpair()
        carrier = control_carrier_from_ipc(local, ledger=ledger)
        sibling = control_carrier_from_ipc(sibling_local, ledger=ledger)
        peer.sendall(b'{"payload":"copy still physically holds this DTO"}\n')
        frame = await carrier.receive_control(maximum_bytes=8192, deadline_ms=boottime_ms()+2000)
        header = frame.header
        credit = ledger.reserve(header.materialization_bytes)
        before = ledger.reserved
        actual_copy = copy.deepcopy
        seen, outcome = {}, {}
        entered, resume = threading.Event(), threading.Event()
        def copy_hook(value, *rest, **kwargs):
            if not entered.is_set():
                entered.set()
                seen['originalDtoPresent'] = value.get('payload') == 'copy still physically holds this DTO'
                if mode == 'reentrant-frame-close':
                    frame.close()
                    seen['heapDuringOriginalDtoCopy'] = ledger.reserved
                else:
                    if not resume.wait(5):
                        raise AssertionError('controlled copy was not released')
            return actual_copy(value, *rest, **kwargs)
        def materialize():
            try:
                outcome['value'] = header.materialize()
            except BaseException as error:
                outcome['error'] = str(error)
        try:
            with patch('mesh_plus.sync.control_carrier.copy.deepcopy', side_effect=copy_hook):
                if mode == 'reentrant-frame-close':
                    materialize()
                else:
                    worker = threading.Thread(target=materialize)
                    worker.start()
                    assert await asyncio.to_thread(entered.wait, 2)
                    close_task = asyncio.create_task(carrier.aclose())
                    try:
                        await asyncio.wait_for(asyncio.shield(close_task), 0.02)
                    except TimeoutError:
                        seen['closeReturnedBeforeCopyUnwind'] = False
                    else:
                        seen['closeReturnedBeforeCopyUnwind'] = True
                    seen['heapDuringOriginalDtoCopy'] = ledger.reserved
                    resume.set()
                    await asyncio.to_thread(worker.join, 2)
                    assert not worker.is_alive()
                    await close_task
            try:
                header.materialization_bytes
            except ProtocolError as error:
                revoked = str(error)
            else:
                revoked = None
            sibling_peer.sendall(b'{}\n')
            other = await sibling.receive_control(maximum_bytes=8192, deadline_ms=boottime_ms()+2000)
            other.close()
            await carrier.aclose()
            await sibling.aclose()
            retained_credit = ledger.reserved
            assert retained_credit == credit.amount
            outcome_value = outcome.pop('value', None)
            returned_after_close = outcome_value is not None
            outcome_value = None
            credit.close()
            counters = [ledger.used, ledger.reserved, ledger.encoded, ledger.encoded_reserved, ledger.pending_units]
            assert counters == [0]*5 and not gc.isenabled()
            cases.append({'mode': mode, 'heapBeforeCopy': before, **seen,
                          'returnedDetachedValueAfterClose': returned_after_close,
                          'headerAfterCloseError': revoked, 'materializeError': outcome.get('error'),
                          'callerCopyCreditRetained': retained_credit, 'finalCounters': counters,
                          'siblingUsable': True})
        finally:
            resume.set()
            frame.close()
            await carrier.aclose()
            await sibling.aclose()
            credit.close()
            peer.close(); sibling_peer.close()
    loaded = {}
    for module in tuple(sys.modules.values()):
        file = getattr(module, '__file__', None)
        if file and Path(file).suffix == '.py' and Path(file).resolve().is_relative_to(args.mesh.resolve()):
            path = Path(file).resolve()
            loaded[str(path.relative_to(args.mesh))] = hashlib.sha256(path.read_bytes()).hexdigest()
    report = {'python': sys.version.split()[0], 'gcDisabled': not gc.isenabled(), 'cases': cases,
              'loadedInputs': loaded, 'scope': 'Public carrier/header/close and ledger APIs; deepcopy hook only controls reentrant/concurrent copy timing. No private source mutation, no measured RSS/cap overrun or late wire output.'}
    args.output.write_text(json.dumps(report, indent=2)+'\n')
    print(json.dumps({'python': report['python'], 'cases': cases, 'loadedInputs': len(loaded)}))

asyncio.run(main())
